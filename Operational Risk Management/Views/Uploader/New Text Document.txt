@functions {
    string GetStatusColorClass(string status)
    {
        return status.ToLower() switch
        {
            "approved" => "bg-emerald-100 text-emerald-800",
            "pending" => "bg-amber-100 text-amber-800",
            "rejected" => "bg-red-100 text-red-800",
            "draft" => "bg-blue-100 text-blue-800",
            _ => "bg-gray-100 text-gray-800"
        };
    }
}
@section Scripts {
    <script>
                   $(document).ready(function() {

          $(".multi-select-btn").on("click", function() {
            const $optionsList = $(this).siblings(".options-list");
            $(".options-list").not($optionsList).hide();
            $optionsList.toggle();
          });

          $(document).on("click", function(event) {
            if (!$(event.target).closest(".multi-select-btn, .options-list").length) {
              $(".options-list").hide();
            }
          });

          function updateMultiSelect(groupClass, hiddenInputId) {
            const selected = [];
            const selectedTexts = [];
            $(`.${groupClass}:checked`).each(function() {
              selected.push($(this).val());
              selectedTexts.push($(this).siblings("label").text());
            });

            $(`#${hiddenInputId}`).val(selected.join(","));
            const $btn = $(`#${hiddenInputId}`).siblings(".multi-select-btn");
            const $text = $btn.find(".selected-text");

            if (selectedTexts.length > 0) {
              $text.text(selectedTexts.join(", "));
            } else {
              const placeholder = $btn.data("placeholder") || "Select options";
              $text.text(placeholder);
            }
          }

          // Initialize on page load
          updateMultiSelect("indicator-checkbox", "hiddenIndicatorIds");
          updateMultiSelect("month-checkbox", "hiddenMonths");
          updateMultiSelect("year-checkbox", "hiddenYears");
          updateMultiSelect("statuses-checkbox", "hiddenStatuses");

          // Bind change events
          $(".indicator-checkbox").on("change", function() {
            updateMultiSelect("indicator-checkbox", "hiddenIndicatorIds");
          });
          $(".month-checkbox").on("change", function() {
            updateMultiSelect("month-checkbox", "hiddenMonths");
          });
          $(".year-checkbox").on("change", function() {
            updateMultiSelect("year-checkbox", "hiddenYears");
          });
          $(".statuses-checkbox").on("change", function() {
            updateMultiSelect("statuses-checkbox", "hiddenStatuses");
          });

        });


    </script>
}
